    SELECT country, revenue, month, year FROM amazon_rev
    UNION ALL
    SELECT 'WOL UK', ROUND(revenue * rate, 2), month, year FROM amazon_rev_wol
    UNION ALL
    SELECT 'Opie Oils UK', ROUND(revenue * rate, 2), month, year FROM amazon_rev_opieoils

    SELECT 'Arnold Clark UK', ROUND(revenue * rate, 2), month, year FROM amazon_rev_arnoldclark

    SELECT 'Retromotion DE', revenue, month, year FROM partner_retromotion

    SELECT 'Iberia WOL', revenue, month, year FROM partner_iberia_wol

      year <> 2026
      OR make_date(year, month, 1) < date_trunc('month', CURRENT_DATE)::date
